    """Spawn isolated Uvicorn server with temp SQLite and cleanup."""

        from app.seed.reset import reset_demo_data

    proc = subprocess.Popen(
        ["python", "-m", "uvicorn", "app.main:app", "--host", "127.0.0.1", "--port", "8001"],
        cwd="backend",
        env=env,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE

    for _ in range(30):
        try:
            response = requests.get("http://127.0.0.1:8001/health")

        yield context

@pytest.fixture

    """Factory to generate JWT tokens for each role."""
    def get_token(username, password):
        response = requests.post(
            f"{app_server['base_url']}/api/v1/auth/login",
            json={"username": username, "password": password}

<threat_model>

## Trust Boundaries

| Boundary | Description |
|-----------|-------------|
| Browser automation → REST/WebSocket APIs | Playwright simulates real user interaction; no API mocking |
| Multi-user sessions → Shared database | Concurrent mutations must not corrupt state |
| UI events → Server-side state | Button clicks trigger API calls; response state must update UI |

## STRIDE Threat Register

| Threat ID | Category | Component | Severity | Disposition | Mitigation Plan |
|-----------|----------|-----------|----------|-------------|-----------------|
| T-05-16 | Spoofing | Multi-user session hijack | low | accept | Tests use seeded credentials; no session fixation tested (API-level testing suffices) |
| T-05-17 | Tampering | Race condition in concurrent confirm | high | mitigate | Stale evaluation detected (409); confirm requires fresh evaluation |
| T-05-18 | Information Disclosure | Denied access HTML | medium | mitigate | Assert denied access page does not render PHI in HTML/JS |
| T-05-19 | Denial of Service | WebSocket infinite reconnect | medium | mitigate | Backoff limits reconnect attempts; timeout prevents busy-loop |

</threat_model>
